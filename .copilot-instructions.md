# Copilot Instructions for OpenDiscourse

## Project Overview

OpenDiscourse is a comprehensive platform for analyzing and processing both media intelligence and government documents. The platform combines advanced data processing capabilities with modern web technologies to deliver intelligent document analysis and retrieval-augmented generation (RAG) capabilities.

## Technology Stack

### Core Technologies
- **Frontend**: Next.js 14+ with App Router, React 18+, TypeScript
- **Backend**: FastAPI (Python 3.11+), PostgreSQL with pgvector/PostGIS
- **Database**: Supabase (self-hosted via Docker), PostgreSQL with vector extensions
- **AI/ML**: LocalAI for inference, Ollama for local LLM agents, LangChain, CrewAI
- **Infrastructure**: Docker, Docker Compose, Ansible for provisioning
- **Data Sources**: govinfo.gov, congress.gov, FRED (Federal Reserve Economic Data)
- **Styling**: Tailwind CSS, Radix UI components
- **State Management**: React Query, Zustand
- **Authentication**: Supabase Auth, Bitwarden CLI integration
- **Testing**: pytest with coverage, structured JSON logging

```python
class JSONFormatter(logging.Formatter):
    def format(self, record):
        log_entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,

# Error handling with retry logic
import asyncio
from functools import wraps

def retry_with_backoff(max_retries: int = 3, backoff_factor: float = 1.0):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_retries - 1:
                        logger.error(
                            f"Function {func.__name__} failed after {max_retries} attempts",
                            extra={'extra_data': {'error': str(e), 'args': args}}
                        )
                        raise

                    wait_time = backoff_factor * (2 ** attempt)
                    logger.warning(
                        f"Attempt {attempt + 1} failed, retrying in {wait_time}s",
                        extra={'extra_data': {'function': func.__name__, 'error': str(e)}}
                    )
                    await asyncio.sleep(wait_time)
        return wrapper
    return decorator
```

```python
# Base ingestion interface
from abc import ABC, abstractmethod
from typing import AsyncGenerator, Dict, Any

class DocumentIngestor(ABC):
    """Base class for document ingestion from various sources"""

    @abstractmethod
    async def fetch_documents(
        self,
        collection: str,
        days_back: int = 30,
        max_documents: int = 100
    ) -> AsyncGenerator[Dict[str, Any], None]:
        """Fetch documents from the source"""
        pass

    @abstractmethod
    async def parse_document(self, raw_doc: Dict[str, Any]) -> DocumentRequest:
        """Parse raw document into standardized format"""
        pass

    @retry_with_backoff(max_retries=3)
    async def ingest_batch(
        self,
        collection: str,
        **kwargs
    ) -> Dict[str, int]:
        """Ingest a batch of documents with error handling"""
        stats = {'processed': 0, 'stored': 0, 'errors': 0}

        async for raw_doc in self.fetch_documents(collection, **kwargs):
            try:
                document = await self.parse_document(raw_doc)
                await self.store_document(document)
                stats['stored'] += 1

                logger.info(
                    f"Successfully ingested document",
                    extra={'extra_data': {
                        'collection': collection,
                        'doc_id': document.title,
                        'source': document.metadata.source
                    }}
                )
            except Exception as e:
                stats['errors'] += 1
                logger.error(
                    f"Failed to process document",
                    extra={'extra_data': {
                        'collection': collection,
                        'error': str(e),
                        'raw_doc_id': raw_doc.get('id', 'unknown')
                    }}
                )
            finally:
                stats['processed'] += 1

        return stats
```

from typing import Protocol
import torch

class DocumentProcessor(Protocol):
    async def process(self, document: DocumentRequest) -> ProcessedDocument:
        ...

class DocumentProcessingPipeline:
    """Complete document processing pipeline"""

    def __init__(self):
        self.entity_extractor = pipeline(
            "ner",
            model="dbmdz/bert-large-cased-finetuned-conll03-english"
        )
        self.embedding_model = SentenceTransformer('all-MiniLM-L6-v2')

    async def process_document(self, document: DocumentRequest) -> ProcessedDocument:
        """Process document through complete pipeline"""

        # 1. Clean and normalize content
        cleaned_content = await self.preprocess_text(document.content)

        # 2. Extract entities
        entities = await self.extract_entities(cleaned_content)

        # 3. Generate embeddings
        embeddings = await self.generate_embeddings(cleaned_content)

        # 4. Detect language
        language = await self.detect_language(cleaned_content)

        return ProcessedDocument(
            id=document.id,
            content=cleaned_content,
            embeddings=embeddings,
            entities=entities,
            metadata=document.metadata,
            language=language
        )

    async def extract_entities(self, content: str) -> Dict[str, List[str]]:
        """Extract named entities with confidence scores"""
        entities = {'PERSON': [], 'ORG': [], 'LOC': [], 'MISC': []}

        # Use async execution for CPU-bound tasks
        loop = asyncio.get_event_loop()
        ner_results = await loop.run_in_executor(
            None,
            self.entity_extractor,
            content
        )

        for entity in ner_results:
            entity_type = entity['entity'].split('-')[-1]
            if entity_type in entities:
                entities[entity_type].append({
                    'text': entity['word'],
                    'confidence': entity['score'],
                    'start': entity['start'],
                    'end': entity['end']
                })

        return entities

##### Database Operations with Async Support
```python
# Async database operations with proper connection management
from sqlalchemy.ext.asyncio import AsyncSession, create_async_engine
from sqlalchemy.orm import sessionmaker
from contextlib import asynccontextmanager

class DatabaseService:
    """Centralized database operations with connection pooling"""

    def __init__(self, database_url: str):
        self.engine = create_async_engine(
            database_url,
            pool_size=20,
            max_overflow=30,
            pool_pre_ping=True,
            echo=False  # Set to True for SQL debugging
        )
        self.async_session = sessionmaker(
            self.engine,
            class_=AsyncSession,
            expire_on_commit=False
        )

    @asynccontextmanager
    async def get_session(self):
        """Context manager for database sessions"""
        async with self.async_session() as session:
            try:
                yield session
                await session.commit()
            except Exception:
                await session.rollback()
                raise
            finally:
                await session.close()

    async def store_document_with_embeddings(
        self,
        document: DocumentRequest,
        embeddings: List[float]
    ) -> UUID:
        """Store document with vector embeddings for search"""
        async with self.get_session() as session:
            db_document = Document(
                title=document.title,
                content=document.content,
                source=document.metadata.source,
                publication_date=document.metadata.publication_date,
                metadata_=document.metadata.dict(),
                embedding=embeddings
            )

            session.add(db_document)
            await session.flush()  # Get the ID

            logger.info(
                "Document stored successfully",
                extra={'extra_data': {
                    'document_id': str(db_document.id),
                    'title': document.title,
                    'source': document.metadata.source
                }}
            )

            return db_document.id

# Vector similarity search using pgvector
async def search_similar_documents(
    query_embedding: List[float],
    limit: int = 10,
    similarity_threshold: float = 0.7
) -> List[Dict[str, Any]]:
    """Search for similar documents using vector similarity"""

    async with database_service.get_session() as session:
        # Using pgvector cosine similarity
        query = text("""
            SELECT
                id, title, content, source, metadata_,
                1 - (embedding <=> :query_embedding) as similarity
            FROM documents
            WHERE 1 - (embedding <=> :query_embedding) > :threshold
            ORDER BY embedding <=> :query_embedding
            LIMIT :limit
        """)

        result = await session.execute(
            query,
            {
                'query_embedding': query_embedding,
                'threshold': similarity_threshold,
                'limit': limit
            }
        )

        return [dict(row) for row in result.fetchall()]
```

# utils/secrets.py - Secure secret management
import subprocess
import json
from typing import Optional, Dict, Any

class BitwardenSecretsManager:
    """Manage secrets using Bitwarden CLI"""

    def __init__(self):
        self.session_key: Optional[str] = None

    async def login(self, email: str, password: str) -> bool:
        """Login to Bitwarden and get session key"""
        try:
            # First, login and get session key
            result = subprocess.run([
                "bw", "login", email, password, "--raw"
            ], capture_output=True, text=True, check=True)

            self.session_key = result.stdout.strip()

            # Unlock the vault
            unlock_result = subprocess.run([
                "bw", "unlock", password, "--raw"
            ], capture_output=True, text=True, check=True)

            self.session_key = unlock_result.stdout.strip()
            return True

        except subprocess.CalledProcessError as e:
            logger.error(f"Bitwarden login failed: {e}")
            return False

    async def get_secret(self, item_name: str) -> Optional[str]:
        """Retrieve secret from Bitwarden vault"""
        if not self.session_key:
            raise ValueError("Not logged in to Bitwarden")

        try:
            result = subprocess.run([
                "bw", "get", "password", item_name, "--session", self.session_key
            ], capture_output=True, text=True, check=True)

            return result.stdout.strip()

        except subprocess.CalledProcessError:
            logger.warning(f"Secret '{item_name}' not found in Bitwarden")
            return None

    async def get_api_keys(self) -> Dict[str, str]:
        """Get all API keys from Bitwarden"""
        api_keys = {}

        key_names = [
            "govinfo-api-key",
            "congress-api-key",
            "fred-api-key",
            "openai-api-key"
        ]

        for key_name in key_names:
            secret = await self.get_secret(key_name)
            if secret:
                api_keys[key_name.replace('-', '_').upper()] = secret

        return api_keys
